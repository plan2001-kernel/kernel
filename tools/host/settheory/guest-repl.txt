# Execute each line at the real Plan2001 MicroPython REPL.
# This test operates only on the disposable QEMU GE/O volume.
import plan9 as p
import settheory as s
v=p.obj_open_volume()
store=s.Store(v)
h1,k1=p.obj_create_ge(v,b'K1')
h2,k2=p.obj_create_ge(v,b'K2')
h3,k3=p.obj_create_ge(v,b'K3')
p.obj_commit(v)
a=store.new((k1,k2))
b=store.new((k2,k3))
assert store.members(a)==tuple(sorted((k1,k2)))
assert store.members(b)==tuple(sorted((k2,k3)))
assert store.intersection(a,b)==(k2,)
assert store.union(a,b)==tuple(sorted((k1,k2,k3)))
assert store.difference(a,b)==(k1,)
assert store.difference(b,a)==(k3,)
assert store.contains(a,k2)
assert not store.contains(a,k3)
assert store.add(a,k3)
assert not store.add(a,k3)
assert store.remove(a,k3)
assert not store.remove(a,k3)
assert store.members(a)==tuple(sorted((k1,k2)))
r=store.materialize(store.intersection(a,b))
assert store.members(r)==(k2,)
assert s.decode(s.encode((k2,k1,k2)))==tuple(sorted((k1,k2)))
p.obj_close(h1)
p.obj_close(h2)
p.obj_close(h3)
p.obj_close(v)
print('SETTHEORY-PASS')
