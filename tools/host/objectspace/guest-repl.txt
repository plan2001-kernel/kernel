# Native MicroPython ObjectSpace graph acceptance, no path or Pgrp.
import plan9 as p
root=p.space_new()
child=p.space_new()
payload=p.obj_create(b'data')
one=p.space_bind(root,'src',payload,p.VREAD+p.VMETA)
two=p.space_bind(root,'src',child,p.VREAD+p.VMETA)
assert one!=two
rows,cursor,epoch=p.space_list(root)
assert len(rows)==2
first,second=rows
bid,label,kind,rights,relation,identity,flags=first
assert bid==one
assert label=='src'
assert rights==p.VREAD+p.VMETA
bid,label,kind,rights,relation,identity,flags=second
assert bid==two
assert kind==p.VKIND_OBJECT_SPACE
reader=p.space_resolve(root,one,p.VREAD)
assert p.obj_read(reader,4)==b'data'
back=p.space_bind(child,'back',root,p.VREAD+p.VMETA)
walk=p.space_resolve(root,two,p.VREAD+p.VMETA)
nested,cursor,nestedepoch=p.space_list(walk)
assert len(nested)==1
alias=p.space_resolve(walk,back,p.VREAD+p.VMETA)
p.obj_close(alias)
p.obj_close(walk)
p.space_unbind(root,one)
assert p.obj_read(reader,4)==b'data'
remaining,cursor,newgen=p.space_list(root)
assert len(remaining)==1
assert newgen!=epoch
p.obj_close(child)
stale,cursor,stalegen=p.space_list(root)
only=next(iter(stale))
bid,label,kind,rights,relation,identity,flags=only
assert flags==1
p.obj_close(reader)
p.obj_close(payload)
p.space_unbind(root,two)
p.obj_close(root)
print('OBJECTSPACE-PASS')
